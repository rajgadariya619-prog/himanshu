        // Mobile Menu Toggle
        const mobileMenuBtn = document.getElementById('mobile-menu-btn');
        const mobileMenu = document.getElementById('mobile-menu');

        mobileMenuBtn.addEventListener('click', () => {
            mobileMenu.classList.toggle('hidden');

        // Portfolio Filtering
        function filterPortfolio(category) {
            const items = document.querySelectorAll('.portfolio-item');
            const tabs = document.querySelectorAll('.portfolio-tab');

            tabs.forEach(tab => {
                if (tab.getAttribute('data-category') === category) {
                    tab.classList.add('bg-gradient-btn', 'text-white');
                    tab.classList.remove('text-gray-400');
                } else {
                    tab.classList.remove('bg-gradient-btn', 'text-white');
                    tab.classList.add('text-gray-400');
                }
            });

            items.forEach(item => {
                if (category === 'all' || item.getAttribute('data-category') === category) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        }

        // Calculator Interactive Logic
        const visitorsSlider = document.getElementById('visitors-slider');
        const revenueSlider = document.getElementById('revenue-slider');
        const nicheSelect = document.getElementById('niche-select');
        const visitorsVal = document.getElementById('visitors-val');
        const revenueVal = document.getElementById('revenue-val');
        const valuationResult = document.getElementById('valuation-result');
        const liftResult = document.getElementById('lift-result');

        function updateCalculator() {
            const visitors = parseInt(visitorsSlider.value);
            const revenue = parseInt(revenueSlider.value);
            const multiple = parseInt(nicheSelect.value);

            visitorsVal.innerText = visitors.toLocaleString() + ' / mo';
            revenueVal.innerText = '$' + revenue.toLocaleString() + ' / mo';

            // Calculate Valuation (Monthly Rev * Multiple)
            const valuation = revenue * multiple;
            valuationResult.innerText = '$' + valuation.toLocaleString();

            // Calculate CRO Revenue Lift (~70% lift)

        // Pricing Toggle Logic
        let isMonthly = false;
        function toggleBilling() {
            isMonthly = !isMonthly;
            const circle = document.getElementById('toggle-circle');
            const prices = document.querySelectorAll('.plan-price');
            const periods = document.querySelectorAll('.plan-period');

            if (isMonthly) {
                circle.classList.add('translate-x-6');
                prices.forEach(price => price.innerText = price.getAttribute('data-monthly'));
                periods.forEach(period => period.innerText = '/ month');
            } else {
                circle.classList.remove('translate-x-6');
                prices.forEach(price => price.innerText = price.getAttribute('data-onetime'));
                periods.forEach(period => period.innerText = '/ project');
            }
        }
